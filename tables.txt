Galerii fotode tabel

CREATE TABLE `if26_karmen_varipuu_id`.`gallery_photos` (`id` INT(4) NOT NULL AUTO_INCREMENT , `filename` VARCHAR(100) NOT NULL , `origname` VARCHAR(100) NOT NULL , `alttext` VARCHAR(256) NULL , `privacy` INT(1) NOT NULL , `userid` INT(4) NOT NULL , `created` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP , `deleted` DATETIME NULL , PRIMARY KEY (`id`)) ENGINE = InnoDB;

Uusima kindla privaatsisega pildi päring

SELECT filename, alttext FROM galleryphotos WHERE id=(SELECT MAX (id) FROM galleryphotos WHERE privacy=3 AND deleted IS NULL);

CREATE TABLE `if26_karmen_varipuu_id`.`users` (`id` INT(4) NOT NULL AUTO_INCREMENT , `firstname` VARCHAR(140) NOT NULL , `lastname` VARCHAR(100) NOT NULL , `birthdate` DATE NOT NULL , `gender` INT(1) NOT NULL , `email` VARCHAR(250) NOT NULL , `password` VARCHAR(120) NOT NULL , `created` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP , PRIMARY KEY (`id`)) ENGINE = InnoDB;
